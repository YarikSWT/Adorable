// Аннуитетный платёж
export function monthly(principal: number, ratePct: number, years: number) {
  const i = ratePct / 100 / 12;
  const n = years * 12;
  return (principal * i * Math.pow(1+i, n)) / (Math.pow(1+i, n) - 1);
}
